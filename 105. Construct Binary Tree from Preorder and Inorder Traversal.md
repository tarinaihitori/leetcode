preorderが根から走査するので、これを基準にできるとわかる。
あとは、inorderから右部分木と左部分木を特定する

```python
class Solution:
    def buildTree(self, preorder: List[int], inorder: List[int]) -> Optional[TreeNode]:
        if not (preorder and inorder):
            return
        root = TreeNode(preorder[0])
        mid = inorder.index(preorder[0])

        root.left = self.buildTree(preorder[1:mid+1], inorder[:mid])
        root.right = self.buildTree(preorder[mid+1:], inorder[mid+1:])
        return root
```

```python
class Solution:
    def buildTree(self, preorder: List[int], inorder: List[int]) -> Optional[TreeNode]:
        value_to_inorder_index = {value:index for index, value in enumerate(inorder)}
        preorder_index = 0

        def build_subtree(left, right):
            nonlocal preorder_index
            if left == right:
                return 
            root_value = preorder[preorder_index]
            inorder_root_index = value_to_inorder_index[root_value]
            root = TreeNode(root_value)

            preorder_index += 1
            root.left = build_subtree(left, inorder_root_index)
            root.right = build_subtree(inorder_root_index + 1, right)
            return root
        return build_subtree(0, len(preorder))
```

```python
class Solution:
    def buildTree(self, preorder, inorder):
        value_to_preorder_index = {val: idx for idx, val in enumerate(preorder)}
        
        stack = []
        
        def gather_descendants(node_position):
            child = None
            while stack:
                if value_to_preorder_index[stack[-1].val] < node_position:
                    break
                back = stack.pop()
                back.right = child
                child = back
            return child
        
        for value in inorder:
            node = TreeNode(value)
            node_position = value_to_preorder_index[node.val]
            node.left = gather_descendants(node_position)
            stack.append(node)
        
        return gather_descendants(-sys.maxsize -1)
```

3rd
再帰でinorderの範囲を渡して木を構築する。

```python
class Solution:
    def buildTree(self, preorder: List[int], inorder: List[int]) -> Optional[TreeNode]:
        value_to_inorder_index = {}
        for index, value in enumerate(inorder):
            value_to_inorder_index[value] = index
        
        preorder_index = 0

        def build_tree_helper(left, right):
            nonlocal preorder_index
            if left == right:
                return 
            root_value = preorder[preorder_index]
            root = TreeNode(root_value)
            root_inorder_index = value_to_inorder_index[root_value]
            preorder_index += 1

            root.left = build_tree_helper(left, root_inorder_index)
            root.right = build_tree_helper(root_inorder_index + 1, right)
            return root
        
        return build_tree_helper(0, len(inorder))
        
```
