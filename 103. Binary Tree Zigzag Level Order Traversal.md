1st
zigzag にするため、boolean の変数で管理する必要あるなあと思った。
あとは、同じ深さごとに処理をするからBFSが素直に感じる。

```python
class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if root is None:
            return []
        queue = deque([root])
        result = []
        left_to_right = True
        while queue:
            same_level_nodes = []
            nodes_count = len(queue)
            for _ in range(nodes_count):
                if left_to_right:
                    node = queue.popleft()
                    same_level_nodes.append(node.val)
                    if node.left:
                        queue.append(node.left)
                    if node.right:
                        queue.append(node.right)
                else:
                    node = queue.pop()
                    same_level_nodes.append(node.val)
                    if node.right:
                        queue.appendleft(node.right)
                    if node.left:
                        queue.appendleft(node.left)
            result.append(same_level_nodes)
            left_to_right = not left_to_right
        return result

```

class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if root is None:
            return []
        result = []
        queue = deque([(root)])
        left_to_right = True
        while queue:
            same_level_nodes = deque()
            for _ in range(len(queue)):
                node = queue.popleft()
                if left_to_right:
                    same_level_nodes.append(node.val)
                else:
                    same_level_nodes.appendleft(node.val)
                if node.left:
                    queue.append(node.left)
                if node.right:
                    queue.append(node.right)
            result.append(list(same_level_nodes))
            left_to_right = not left_to_right
        return result

```python
class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:
            return []
        result = []
        level = 0
        nodes = [root]
        
        while nodes:
            current_level_values = []
            next_nodes = []
            for node in nodes:
                current_level_values.append(node.val)
                if node.left:
                    next_nodes.append(node.left)
                if node.right:
                    next_nodes.append(node.right)
            if level % 2 == 1:
                current_level_values.reverse()
            nodes = next_nodes
            result.append(current_level_values)
            level += 1
        return result
```

再帰的な方法も一応。

```python
class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        result = []

        def traverse_levels(node, level):
            if node is None:
                return
            if level >= len(result):
                result.append(deque())
            if level % 2 == 0:
                result[level].append(node.val)
            else:
                result[level].appendleft(node.val)
            traverse_levels(node.left, level + 1)
            traverse_levels(node.right, level + 1)

        traverse_levels(root, 0)
        return [list(values) for values in result]
```
