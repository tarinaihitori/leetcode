```python
# TLEしたコード
class Solution:
    def maxSubArray(self, nums: List[int]) -> int:
        if not nums:
            return 0

        num_length = len(nums)
        max_sum =nums[0]

        for i in range(num_length):
            for j in range(i, num_length):
                max_sum = max(max_sum, sum(nums[i:j+1]))
        return max_sum
```
